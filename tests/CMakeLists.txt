# One Qt Test executable per test file, registered with CTest.
function(ppcb_add_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE pocketpcb_core Qt6::Test)
    add_test(NAME ${name} COMMAND ${name})
endfunction()

# The same for tests that need the GUI library; they run without a display.
function(ppcb_add_gui_test name)
    ppcb_add_test(${name})
    target_link_libraries(${name} PRIVATE pocketpcb_gui)
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()
